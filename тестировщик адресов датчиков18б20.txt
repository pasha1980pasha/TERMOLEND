// ========== НАСТРОЙКИ ==========
const char* ssid = "TERMO_LABORATORY";
const char* password = "www111www";
#define DS18B20_PIN 4

WebServer server(80);
OneWire oneWire(DS18B20_PIN);

byte foundAddr[8];
bool sensorFound = false;
String sensorAddress = "";
String byteArrayString = "";

// ========== ПОИСК ДАТЧИКА ==========
void scanForSensors() {
  Serial.println("\n🔍 Scanning DS18B20 on GPIO" + String(DS18B20_PIN) + "...");
  
  byte addr[8];
  oneWire.reset_search();
  sensorFound = false;
  
  while (oneWire.search(addr)) {
    if (OneWire::crc8(addr, 7) == addr[7]) {
      sensorFound = true;
      memcpy(foundAddr, addr, 8);
      
      // Адрес в виде строки
      sensorAddress = "";
      for (int i = 0; i < 8; i++) {
        if (foundAddr[i] < 0x10) sensorAddress += "0";
        sensorAddress += String(foundAddr[i], HEX);
        if (i < 7) sensorAddress += ":";
      }
      
      // Формат для вставки в код
      byteArrayString = "";
      for (int i = 0; i < 8; i++) {
        if (i > 0) byteArrayString += ", ";
        byteArrayString += "0x" + String(foundAddr[i], HEX);
      }
      
      Serial.print("✅ FOUND! Address: ");
      Serial.println(sensorAddress);
      Serial.print("   Copy: byte addr[8] = {");
      Serial.print(byteArrayString);
      Serial.println("};");
      
      return;
    }
  }
  
  if (!sensorFound) {
    Serial.println("❌ NO SENSOR!");
  }
}

// ========== HTTP ==========
void handleRoot() {
  server.send_P(200, "text/html", index_html);
}

void handleData() {
  scanForSensors();
  
  String json = "{";
  json += "\"found\":" + String(sensorFound ? "true" : "false") + ",";
  json += "\"address\":\"" + sensorAddress + "\",";
  json += "\"byteArray\":\"" + byteArrayString + "\",";
  json += "\"pin\":" + String(DS18B20_PIN);
  json += "}";
  
  server.send(200, "application/json", json);
}

// ========== SETUP ==========
void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("\n=================================");
  Serial.println("DS18B20 SCANNER on GPIO" + String(DS18B20_PIN));
  Serial.println("=================================");
  
  scanForSensors();
  
  WiFi.mode(WIFI_AP);
  WiFi.softAP(ssid, password);
  WiFi.softAPConfig(IPAddress(10, 0, 0, 1), IPAddress(10, 0, 0, 1), IPAddress(255, 255, 255, 0));
  
  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.begin();
  
  Serial.print("WiFi AP: ");
  Serial.println(ssid);
  Serial.print("IP: http://");
  Serial.println(WiFi.softAPIP());
  Serial.println("=================================\n");
}

void loop() {
  server.handleClient();
  delay(10);
