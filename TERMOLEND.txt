String addrToString(byte addr[8]) {
  String str = "";
  for (int i = 0; i < 8; i++) {
    if (addr[i] < 0x10) str += "0";
    str += String(addr[i], HEX);
    if (i < 7) str += ":";
  }
  return str;
}

bool hasEnoughFreeSpace() {
  if (!fsMounted) return false;
  
  unsigned long totalBytes = LittleFS.totalBytes();
  unsigned long usedBytes = LittleFS.usedBytes();
  unsigned long freeBytes = totalBytes - usedBytes;
  
  if (freeBytes < 50 * 1024) {
    Serial.printf("WARNING: Low on space! Free: %lu bytes\n", freeBytes);
    return false;
  }
  return true;
}

void transferLastLines(const char* sourceFile, const char* destFile, int linesToKeep) {
  File src = LittleFS.open(sourceFile, "r");
  if (!src) return;
  
  int totalLines = 0;
  while (src.available()) {
    if (src.readStringUntil('\n').length() > 0) {
      totalLines++;
    }
  }
  src.seek(0);
  
  int startLine = (totalLines > linesToKeep) ? (totalLines - linesToKeep) : 0;
  
  File dst = LittleFS.open(destFile, "w");
  if (!dst) {
    src.close();
    return;
  }
  
  int currentLine = 0;
  while (src.available()) {
    String line = src.readStringUntil('\n');
    if (line.length() > 0) {
      if (currentLine >= startLine) {
        dst.println(line);
      }
      currentLine++;
    }
  }
  
  src.close();
  dst.close();
  Serial.printf("Transferred %d lines from backup\n", currentLine - startLine);
}

void rotateHistoryFile() {
  if (!fsMounted) return;
  
  File file = LittleFS.open("/history.csv", "r");
  if (!file) return;
  
  unsigned long fileSize = file.size();
  file.close();
  
  if (fileSize > MAX_HISTORY_FILE_SIZE) {
    Serial.printf("History file too large (%lu bytes), rotating...\n", fileSize);
    
    if (LittleFS.exists(HISTORY_BACKUP_FILE)) {
      LittleFS.remove(HISTORY_BACKUP_FILE);
    }
    
    if (LittleFS.rename("/history.csv", HISTORY_BACKUP_FILE)) {
      Serial.println("History file rotated successfully");
      
      File newFile = LittleFS.open("/history.csv", "w");
      if (newFile) {
        newFile.close();
      }
      
      transferLastLines(HISTORY_BACKUP_FILE, "/history.csv", 100);
      
    } else {
      Serial.println("Failed to rotate history file!");
    }
  }
}

void loadHistoryFromFlash() {
  if (!fsMounted) return;
  
  // Сначала пробуем загрузить из основного файла
  File file = LittleFS.open("/history.csv", "r");
  if (!file) {
    // Если нет, пробуем из бэкапа
    file = LittleFS.open(HISTORY_BACKUP_FILE, "r");
    if (!file) {
      Serial.println("No history file found, starting fresh");
      return;
    }
    Serial.println("Loading from backup file");
  }
  
  int loadedCount = 0;
  String line;
  unsigned long lastTimestamp = 0;
  
  for (int i = 0; i < MAX_SENSORS; i++) {
    sensorsList[i].histIndex = 0;
    sensorsList[i].historyCount = 0;
  }
  
  while (file.available() && loadedCount < MAX_HISTORY) {
    line = file.readStringUntil('\n');
    line.trim();
    if (line.length() == 0) continue;
    
    int comma1 = line.indexOf(',');
    int comma2 = line.indexOf(',', comma1 + 1);
    int comma3 = line.indexOf(',', comma2 + 1);
    int comma4 = line.indexOf(',', comma3 + 1);
    
    if (comma1 > 0 && comma4 > 0) {
      unsigned long timestamp = line.substring(0, comma1).toInt();
      float t0 = line.substring(comma1 + 1, comma2).toFloat();
      float t1 = line.substring(comma2 + 1, comma3).toFloat();
      float t2 = line.substring(comma3 + 1, comma4).toFloat();
      float t3 = line.substring(comma4 + 1).toFloat();
      
      if (sensorsList[0].historyCount < MAX_HISTORY) {
        sensorsList[0].history[sensorsList[0].historyCount] = t0;
        sensorsList[0].historyCount++;
      }
      if (sensorsList[1].historyCount < MAX_HISTORY) {
        sensorsList[1].history[sensorsList[1].historyCount] = t1;
        sensorsList[1].historyCount++;
      }
      if (sensorsList[2].historyCount < MAX_HISTORY) {
        sensorsList[2].history[sensorsList[2].historyCount] = t2;
        sensorsList[2].historyCount++;
      }
      if (sensorsList[3].historyCount < MAX_HISTORY) {
        sensorsList[3].history[sensorsList[3].historyCount] = t3;
        sensorsList[3].historyCount++;
      }
      loadedCount++;
      lastTimestamp = timestamp;
    }
  }
  file.close();
  
  for (int i = 0; i < MAX_SENSORS; i++) {
    sensorsList[i].histIndex = sensorsList[i].historyCount;
  }
  
  if (lastTimestamp > 0) {
    totalUptime = lastTimestamp;
  }
  
  Serial.printf("Loaded %d points from Flash\n", loadedCount);
}

// ============================================================================
void flushBufferToFlash() {
  if (!fsMounted) return;
  if (flashBuffer.index == 0) return;
  
  // Проверяем свободное место перед записью
  if (!hasEnoughFreeSpace()) {
    Serial.println("Low space, forcing rotation before flush...");
    rotateHistoryFile();
  }
  
  File file = LittleFS.open("/history.csv", "a");
  if (!file) {
    Serial.println("ERROR: Failed to open history file for writing");
    file = LittleFS.open("/history.csv", "w");
    if (!file) {
      Serial.println("CRITICAL: Cannot write to Flash!");
      return;
    }
  }
  
  for (int i = 0; i < flashBuffer.index; i++) {
    file.printf("%lu,%.1f,%.1f,%.1f,%.1f\n",
      flashBuffer.timestamps[i],
      flashBuffer.temps[0][i],
      flashBuffer.temps[1][i],
      flashBuffer.temps[2][i],
      flashBuffer.temps[3][i]);
  }
  file.close();
  
  Serial.printf("Saved %d points to Flash\n", flashBuffer.index);
  flashBuffer.index = 0;
  
  // Проверяем размер файла после записи
  rotateHistoryFile();
}

void addToFlashBuffer(float temps[4], unsigned long timestamp) {
  if (!fsMounted) return;
  
  for (int s = 0; s < MAX_SENSORS; s++) {
    flashBuffer.temps[s][flashBuffer.index] = temps[s];
  }
  flashBuffer.timestamps[flashBuffer.index] = timestamp;
  flashBuffer.index++;
  
  if (flashBuffer.index >= CSV_BUFFER_SIZE) {
    flushBufferToFlash();
  }
}

void readAllTemperatures() {
  sensors.requestTemperatures();
  
  unsigned long startWait = millis();
  while ((millis() - startWait) < 750) {
    server.handleClient();
    delay(10);
  }
  
  float currentTemps[4];
  unsigned long now = getRealTimestamp();
  
  for (int i = 0; i < MAX_SENSORS; i++) {
    float temp = sensors.getTempC(sensorsList[i].addr);
    
    if (temp == DEVICE_DISCONNECTED_C) {
      sensorsList[i].currentTemp = -127;
      currentTemps[i] = -127;
      Serial.printf("Sensor %d (%s) DISCONNECTED\n", i, sensorsList[i].name.c_str());
      continue;
    }
    
    if (!isnan(temp) && temp > -50 && temp < 100) {
      sensorsList[i].currentTemp = temp;
      currentTemps[i] = temp;
      
      if (sensorsList[i].enabled) {
        if (sensorsList[i].historyCount < MAX_HISTORY) {
          sensorsList[i].history[sensorsList[i].historyCount] = temp;
          sensorsList[i].historyCount++;
          sensorsList[i].histIndex = sensorsList[i].historyCount;
        } else {
          sensorsList[i].history[sensorsList[i].histIndex % MAX_HISTORY] = temp;
          sensorsList[i].histIndex++;
        }
      }
      
      Serial.printf("sensor%d:%.2f|g\n", i, temp);
      
    } else {
      sensorsList[i].currentTemp = -127;
      currentTemps[i] = -127;
      Serial.printf("Sensor %d (%s) INVALID: %.2f\n", i, sensorsList[i].name.c_str(), temp);
    }
  }
  
  Serial.println("---");
  addToFlashBuffer(currentTemps, now);
}

String getHistoryForSensor(int id) {
  String data = "[";
  int count = sensorsList[id].historyCount;
  
  if (count == 0) {
    data += "]";
    return data;
  }
  
  int step = 1;
  if (count > 800) step = count / 800;
  if (step < 1) step = 1;
  
  for (int j = 0; j < count; j += step) {
    if (j > 0) data += ",";
    data += String(sensorsList[id].history[j], 1);
  }
  data += "]";
  return data;
}

function initChart() {
  const canvas = document.getElementById('tempChart');
  canvas.width = canvas.clientWidth;
  canvas.height = canvas.clientHeight;
  chart = {
    canvas: canvas,

function drawChart() {
  if (!chart) return;
  const ctx = chart.ctx;
  const w = chart.width;
  const h = chart.height;
  const p = chart.padding;
  ctx.clearRect(0, 0, w, h);
  ctx.fillStyle = 'white';
  ctx.fillRect(0, 0, w, h);
  
  let allData = [];
  for (let id of activeSensors) {
    if (historyData[id] && historyData[id].length > 0) {
      allData = allData.concat(historyData[id]);
    }
  }
  
  if (allData.length === 0 || activeSensors.length === 0) {
    ctx.fillStyle = '#999';
    ctx.font = '14px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('Select sensors to display', w/2, h/2);
    return;
  }
  
  let minTemp = Math.min(...allData);
  let maxTemp = Math.max(...allData);
  let maxLen = 0;
  for (let id of activeSensors) {
    if (historyData[id] && historyData[id].length > maxLen) {
      maxLen = historyData[id].length;
    }
  }
  
  totalPoints = maxLen;
  let minTime = zoomMin;
  let maxTime = zoomMax;
  if (maxTime > maxLen) {
    maxTime = maxLen;
  }
  if (minTime >= maxTime) {
    minTime = 0;
    maxTime = maxLen;
  }
  
  const timeRange = maxTime - minTime;
  if (timeRange <= 0) return;
  
  const tempRange = maxTemp - minTemp;
  const yMin = minTemp - (tempRange * 0.1);
  const yMax = maxTemp + (tempRange * 0.1);
  const cw = w - p.left - p.right;
  const ch = h - p.top - p.bottom;
  
  ctx.strokeStyle = '#e0e0e0';
  ctx.lineWidth = 1;
  ctx.setLineDash([5, 5]);
  for (let i = 0; i <= 4; i++) {
    const y = p.top + (ch * i / 4);
    ctx.beginPath();
    ctx.moveTo(p.left, y);
    ctx.lineTo(w - p.right, y);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = '#666';
    ctx.font = '10px Arial';
    ctx.textAlign = 'right';
    const value = yMax - ((yMax - yMin) * i / 4);
    ctx.fillText(value.toFixed(1) + '°C', p.left - 5, y + 3);
  }
  
  ctx.setLineDash([]);
  for (let idx = 0; idx < activeSensors.length; idx++) {
    const id = activeSensors[idx];
    const data = historyData[id];
    if (!data || data.length < 2) continue;
    
    ctx.beginPath();
    ctx.strokeStyle = colors[idx % colors.length];
    ctx.lineWidth = 2.5;
    let firstPoint = true;
    
    for (let i = Math.floor(minTime); i <= Math.floor(maxTime) && i < data.length; i++) {
      const x = p.left + ((i - minTime) / timeRange) * cw;
      const y = p.top + ch - ((data[i] - yMin) / (yMax - yMin)) * ch;
      if (firstPoint) {
        ctx.moveTo(x, y);
        firstPoint = false;
      } else {
        ctx.lineTo(x, y);
      }
    }
    ctx.stroke();
  }
  
  for (let i = 0; i <= 4; i++) {
    const timePos = minTime + (timeRange * i / 4);
    const x = p.left + (i / 4) * cw;
    const secondsAgo = Math.round((maxLen - timePos) * currentInterval);
    let label = '';
    if (secondsAgo < 60) {
      label = secondsAgo + 's';
    } else if (secondsAgo < 3600) {
      label = Math.floor(secondsAgo / 60) + 'm';
    } else if (secondsAgo < 86400) {
      label = Math.floor(secondsAgo / 3600) + 'h';
    } else {
      label = Math.floor(secondsAgo / 86400) + 'd';
    }
    ctx.fillStyle = '#888';
    ctx.font = '9px Arial';
    ctx.textAlign = 'center';
    ctx.fillText(label, x, h - p.bottom + 12);
  }
  
  const hoursAvailable = (maxLen * currentInterval / 3600).toFixed(1);
  document.getElementById('hoursInfo').innerHTML = `📊 History: ${maxLen} points | ${hoursAvailable} hours (${(hoursAvailable/24).toFixed(1)} days)`;
  document.getElementById('status').innerHTML = `✅ Updated: ${new Date().toLocaleTimeString()} | Points: ${maxLen}`;
}

function handleMouseDown(e) {
  const canvas = document.getElementById('tempChart');
  const rect = canvas.getBoundingClientRect();
  const scaleX = canvas.width / rect.width;
  const x = (e.clientX - rect.left) * scaleX;
  const p = chart.padding;
  if (x >= p.left && x <= canvas.width - p.right) {
    isDragging = true;
    dragStartX = x;
    dragStartMin = zoomMin;
    dragStartMax = zoomMax;
    canvas.style.cursor = 'grabbing';
  }
}

function handleMouseMove(e) {
  if (!isDragging) return;
  const canvas = document.getElementById('tempChart');
  const rect = canvas.getBoundingClientRect();
  const scaleX = canvas.width / rect.width;
  const x = (e.clientX - rect.left) * scaleX;
  const p = chart.padding;
  
  let maxLen = 0;
  for (let id of activeSensors) {
    if (historyData[id] && historyData[id].length > maxLen) {
      maxLen = historyData[id].length;
    }
  }
  if (maxLen === 0) return;
  
  const deltaX = x - dragStartX;
  const currentRange = dragStartMax - dragStartMin;
  if (currentRange <= 0) return;
  
  const deltaPoints = (deltaX / (canvas.width - p.left - p.right)) * currentRange;
  let newMin = dragStartMin - deltaPoints;
  let newMax = dragStartMax - deltaPoints;
  
  if (newMin < 0) {
    newMin = 0;
    newMax = currentRange;
  }
  if (newMax > maxLen) {
    newMax = maxLen;
    newMin = maxLen - currentRange;
  }
  if (newMin >= 0 && newMax <= maxLen && (newMax - newMin) > 5) {
    zoomMin = Math.floor(newMin);
    zoomMax = Math.ceil(newMax);
    drawChart();
  }
}

function handleMouseUp() {
  isDragging = false;
  const canvas = document.getElementById('tempChart');
  canvas.style.cursor = 'default';
}

function handleWheel(e) {

      maxLen = historyData[id].length;

  if (currentMax > maxLen) currentMax = maxLen;

    drawChart();

  Serial.println("🌡️ GREENHOUSE MONITOR v4.0");

unsigned long lastRead = 0;

void loop() {
  unsigned long currentMillis = millis();
  
  // Temperature reading with interval
  if (currentMillis - lastRead >= (unsigned long)currentInterval * 1000) {
    lastRead = currentMillis;
    readAllTemperatures();
  }
  
  // Auto-flush buffer every 5 minutes
  if (fsMounted && flashBuffer.index > 0 && (currentMillis - lastAutoFlush >= AUTO_FLUSH_INTERVAL)) {
    lastAutoFlush = currentMillis;
    flushBufferToFlash();
  }
  
  // Save system time every minute
  if (fsMounted && (currentMillis - lastSavedTime >= 60000)) {
    saveSystemTime();
  }
  
  // Check file size every hour
  static unsigned long lastRotationCheck = 0;
  if (fsMounted && (currentMillis - lastRotationCheck >= 3600000)) {
    lastRotationCheck = currentMillis;
    rotateHistoryFile();
  }
  
  // Handle web clients
  server.handleClient();
  delay(10);
}
