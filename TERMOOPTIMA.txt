String addrToString(byte addr[8]) {
  String str = "";
  for (int i = 0; i < 8; i++) {
    if (addr[i] < 0x10) str += "0";
    str += String(addr[i], HEX);
    if (i < 7) str += ":";
  }
  return str;
}

bool hasEnoughFreeSpace() {
  if (!fsMounted) return false;
  unsigned long freeBytes = LittleFS.totalBytes() - LittleFS.usedBytes();
  if (freeBytes < 50 * 1024) {
    Serial.printf("WARNING: Low on space! Free: %lu bytes\n", freeBytes);
    return false;
  }
  return true;
}

void transferLastLines(const char* sourceFile, const char* destFile, int linesToKeep) {
  File src = LittleFS.open(sourceFile, "r");
  if (!src) return;
  int totalLines = 0;
  while (src.available()) if (src.readStringUntil('\n').length() > 0) totalLines++;
  src.seek(0);
  int startLine = (totalLines > linesToKeep) ? (totalLines - linesToKeep) : 0;
  File dst = LittleFS.open(destFile, "w");
  if (!dst) { src.close(); return; }
  int currentLine = 0;
  while (src.available()) {
    String line = src.readStringUntil('\n');
    if (line.length() > 0 && currentLine >= startLine) dst.println(line);
    currentLine++;
  }
  src.close(); dst.close();
}

void rotateHistoryFile() {
  if (!fsMounted) return;
  File file = LittleFS.open("/history.csv", "r");
  if (!file) return;
  unsigned long fileSize = file.size();
  file.close();
  if (fileSize > MAX_HISTORY_FILE_SIZE) {
    Serial.printf("History file too large (%lu bytes), rotating...\n", fileSize);
    if (LittleFS.exists(HISTORY_BACKUP_FILE)) LittleFS.remove(HISTORY_BACKUP_FILE);
    if (LittleFS.rename("/history.csv", HISTORY_BACKUP_FILE)) {
      Serial.println("History file rotated successfully");
      File newFile = LittleFS.open("/history.csv", "w");
      if (newFile) newFile.close();
      transferLastLines(HISTORY_BACKUP_FILE, "/history.csv", 100);
    }
  }
}

void loadHistoryFromFlash() {
  if (!fsMounted) return;
  File file = LittleFS.open("/history.csv", "r");
  if (!file) {
    file = LittleFS.open(HISTORY_BACKUP_FILE, "r");
    if (!file) { Serial.println("No history file found, starting fresh"); return; }
    Serial.println("Loading from backup file");
  }
  int loadedCount = 0;
  String line;
  unsigned long lastTimestamp = 0;
  for (int i = 0; i < MAX_SENSORS; i++) {
    sensorsList[i].histIndex = 0;
    sensorsList[i].historyCount = 0;
  }
  while (file.available() && loadedCount < MAX_HISTORY) {
    line = file.readStringUntil('\n');
    line.trim();
    if (line.length() == 0) continue;
    int comma1 = line.indexOf(',');
    int comma2 = line.indexOf(',', comma1 + 1);
    int comma3 = line.indexOf(',', comma2 + 1);
    int comma4 = line.indexOf(',', comma3 + 1);
    if (comma1 > 0 && comma4 > 0) {
      unsigned long timestamp = line.substring(0, comma1).toInt();
      float t0 = line.substring(comma1 + 1, comma2).toFloat();
      float t1 = line.substring(comma2 + 1, comma3).toFloat();
      float t2 = line.substring(comma3 + 1, comma4).toFloat();
      float t3 = line.substring(comma4 + 1).toFloat();
      if (sensorsList[0].historyCount < MAX_HISTORY) { sensorsList[0].history[sensorsList[0].historyCount] = t0; sensorsList[0].historyCount++; }
      if (sensorsList[1].historyCount < MAX_HISTORY) { sensorsList[1].history[sensorsList[1].historyCount] = t1; sensorsList[1].historyCount++; }
      if (sensorsList[2].historyCount < MAX_HISTORY) { sensorsList[2].history[sensorsList[2].historyCount] = t2; sensorsList[2].historyCount++; }
      if (sensorsList[3].historyCount < MAX_HISTORY) { sensorsList[3].history[sensorsList[3].historyCount] = t3; sensorsList[3].historyCount++; }
      loadedCount++;
      lastTimestamp = timestamp;
    }
  }
  file.close();
  for (int i = 0; i < MAX_SENSORS; i++) sensorsList[i].histIndex = sensorsList[i].historyCount;
  if (lastTimestamp > 0) totalUptime = lastTimestamp;
  Serial.printf("Loaded %d points from Flash\n", loadedCount);
}

// ЗАПИСЬ БУФЕРА В FLASH
// ============================================================================
void flushBufferToFlash() {
  if (!fsMounted) return;
  if (flashBuffer.index == 0) return;
  if (!hasEnoughFreeSpace()) rotateHistoryFile();
  File file = LittleFS.open("/history.csv", "a");
  if (!file) {
    file = LittleFS.open("/history.csv", "w");
    if (!file) { Serial.println("CRITICAL: Cannot write to Flash!"); return; }
  }
  for (int i = 0; i < flashBuffer.index; i++) {
    file.printf("%lu,%.1f,%.1f,%.1f,%.1f\n",
      flashBuffer.timestamps[i],
      flashBuffer.temps[0][i],
      flashBuffer.temps[1][i],
      flashBuffer.temps[2][i],
      flashBuffer.temps[3][i]);
  }
  file.close();
  Serial.printf("Saved %d points to Flash\n", flashBuffer.index);
  flashBuffer.index = 0;
  rotateHistoryFile();
}

void addToFlashBuffer(float temps[4], unsigned long timestamp) {
  if (!fsMounted) return;
  for (int s = 0; s < MAX_SENSORS; s++) flashBuffer.temps[s][flashBuffer.index] = temps[s];
  flashBuffer.timestamps[flashBuffer.index] = timestamp;
  flashBuffer.index++;
  if (flashBuffer.index >= CSV_BUFFER_SIZE) flushBufferToFlash();
}

void readAllTemperatures() {
  sensors.requestTemperatures();
  unsigned long startWait = millis();
  while ((millis() - startWait) < 750) { server.handleClient(); delay(10); }
  float currentTemps[4];
  unsigned long now = getRealTimestamp();
  for (int i = 0; i < MAX_SENSORS; i++) {
    float temp = sensors.getTempC(sensorsList[i].addr);
    if (temp == DEVICE_DISCONNECTED_C) {
      sensorsList[i].currentTemp = -127;
      currentTemps[i] = -127;
      Serial.printf("Sensor %d (%s) DISCONNECTED\n", i, sensorsList[i].name.c_str());
      continue;
    }
    if (!isnan(temp) && temp > -50 && temp < 100) {
      sensorsList[i].currentTemp = temp;
      currentTemps[i] = temp;
      if (sensorsList[i].enabled) {
        if (sensorsList[i].historyCount < MAX_HISTORY) {
          sensorsList[i].history[sensorsList[i].historyCount] = temp;
          sensorsList[i].historyCount++;
          sensorsList[i].histIndex = sensorsList[i].historyCount;
        } else {
          sensorsList[i].history[sensorsList[i].histIndex % MAX_HISTORY] = temp;
          sensorsList[i].histIndex++;
        }
      }
      Serial.printf("sensor%d:%.2f|g\n", i, temp);
    } else {
      sensorsList[i].currentTemp = -127;
      currentTemps[i] = -127;
      Serial.printf("Sensor %d (%s) INVALID: %.2f\n", i, sensorsList[i].name.c_str(), temp);
    }
  }
  Serial.println("---");
  addToFlashBuffer(currentTemps, now);
}

String getHistoryForSensor(int id) {
  String data = "[";
  int count = sensorsList[id].historyCount;
  if (count == 0) { data += "]"; return data; }
  int step = 1;
  if (count > 800) step = count / 800;
  if (step < 1) step = 1;
  for (int j = 0; j < count; j += step) {
    if (j > 0) data += ",";
    data += String(sensorsList[id].history[j], 1);
  }
  data += "]";
  return data;
}

unsigned long lastRead = 0;
void loop() {
  unsigned long now = millis();
  if (now - lastRead >= (unsigned long)currentInterval * 1000) { lastRead = now; readAllTemperatures(); }
  if (fsMounted && flashBuffer.index > 0 && (now - lastAutoFlush >= AUTO_FLUSH_INTERVAL)) { lastAutoFlush = now; flushBufferToFlash(); }
  if (fsMounted && (now - lastSavedTime >= 60000)) saveSystemTime();
  static unsigned long lastRot = 0; if (fsMounted && (now - lastRot >= 3600000)) { lastRot = now; rotateHistoryFile(); }
  server.handleClient(); delay(10);
}
